% Title row of an entry: title on the left, date or link on the right.
% Same layout as twocolentry, but built as a table instead of with paracol:
% paracol forces its own page breaks, which could split a title from its
% bullets. A table row is never split, so it stays with what follows.
\newenvironment{entrytitle}[1]{%
    \def\entrySecondColumn{#1}%
    \par\noindent
    \tabularx{\linewidth}{@{}>{\raggedright\arraybackslash}X@{\hspace{\columnsep}}>{\raggedleft\arraybackslash}p{4.5 cm}@{}}%
}{%
    & \entrySecondColumn
    \endtabularx\par
}

% Keeps one whole entry (title, tools line, bullets) on the same page.
% Inside it every page break is forbidden, so LaTeX can only break between
% entries: if an entry doesn't fit, all of it moves to the next page.
\makeatletter
\newenvironment{keeptogether}{
    \par
    \interlinepenalty=10000 \clubpenalty=10000 \widowpenalty=10000
    \@beginparpenalty=10000 \@itempenalty=10000 \@endparpenalty=10000
}{
    \par
    % Allow a page break again right after the entry. Use the same cost as the
    % break before a section heading, so LaTeX fills the page instead of
    % preferring to break early at a heading.
    \penalty\@secpenalty
}
\makeatother

\newenvironment{twocolentry}[2][]{
    \onecolentry
    \def\secondColumn{#2}
    \setcolumnwidth{\fill, 4.5 cm}
    \begin{paracol}{2}
}{
    \switchcolumn \raggedleft \secondColumn
    \end{paracol}
    \endonecolentry
}

\newenvironment{threecolentry}[3][]{
    \onecolentry
    \def\thirdColumn{#3}
    \setcolumnwidth{, \fill, 4.5 cm}
    \begin{paracol}{3}
    {\raggedright #2} \switchcolumn
}{
    \switchcolumn \raggedleft \thirdColumn
    \end{paracol}
    \endonecolentry
}
